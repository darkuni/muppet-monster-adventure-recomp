cmake_minimum_required(VERSION 3.16)
project(muppetmonster CXX)
if(WIN32)
  enable_language(RC)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
file(GLOB GEN_SRC CONFIGURE_DEPENDS gen/*.cpp)
file(GLOB RT_SRC CONFIGURE_DEPENDS runtime/*.cpp)
add_executable(muppetmonster ${RT_SRC} ${GEN_SRC})
target_include_directories(muppetmonster PRIVATE runtime gen)
# The generated code is large and simple: -O1 compiles much faster with little speed loss.
set_source_files_properties(${GEN_SRC} PROPERTIES COMPILE_OPTIONS "-O1;-fno-strict-aliasing;-w;-g1")
if(WIN32)
  set(SDL2_DIR "" CACHE PATH "SDL2 mingw development package (x86_64-w64-mingw32 folder)")
  target_include_directories(muppetmonster PRIVATE runtime/gl ${SDL2_DIR}/include)
  target_compile_definitions(muppetmonster PRIVATE SDL_MAIN_HANDLED)
  target_compile_options(muppetmonster PRIVATE -O2 -fno-strict-aliasing -g0)
  target_link_directories(muppetmonster PRIVATE ${SDL2_DIR}/lib)
  if(EXISTS ${CMAKE_SOURCE_DIR}/runtime/app.rc)
    target_sources(muppetmonster PRIVATE runtime/app.rc)
  endif()
  target_link_libraries(muppetmonster PRIVATE SDL2 winmm -mwindows -static-libgcc -static-libstdc++ -Wl,-Bstatic -lwinpthread -Wl,-Bdynamic)
elseif(USE_SDL)
  target_compile_definitions(muppetmonster PRIVATE USE_SDL)
  target_compile_options(muppetmonster PRIVATE -O2 -fno-strict-aliasing -g)
  target_link_libraries(muppetmonster PRIVATE SDL2 GL pthread)
else()
  target_compile_options(muppetmonster PRIVATE -O2 -fno-strict-aliasing -g)
  target_link_libraries(muppetmonster PRIVATE EGL GL pthread)
endif()
